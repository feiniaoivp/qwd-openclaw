# 可嵌入 daily_watch.py 的汇川技术专用监控逻辑
def check_huichuan_signal(df):
    close = df['close']
    ma20 = close.rolling(20).mean()
    bb_upper = ma20 + 2*close.rolling(20).std()
    bb_lower = ma20 - 2*close.rolling(20).std()
    vol = df['volume']
    vol_ma20 = vol.rolling(20).mean()
    
    c = close.iloc[-1]
    signals = []
    
    # 买入信号
    if c > bb_upper.iloc[-1] and vol.iloc[-1] > vol_ma20.iloc[-1] * 1.5:
        signals.append(("BUY", "突破布林上轨+放量", c))
    if abs(c - ma20.iloc[-1]) / ma20.iloc[-1] < 0.01 and vol.iloc[-1] > vol_ma20.iloc[-1]:
        signals.append(("BUY_WEAK", "回踩MA20企稳+量能", c))
    
    # 止损信号
    if c < bb_lower.iloc[-1]:
        signals.append(("STOP_LOSS", "跌破布林下轨", c))
    if c < 48.50:
        signals.append(("HARD_STOP", "破年内低点硬止损", c))
    
    return signals
